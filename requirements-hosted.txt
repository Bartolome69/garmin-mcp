# Pinned dependencies for the hosted server image.
#
# Separate from requirements.txt on purpose. That file stays unpinned so the
# local installer can fall back to constraints-legacy.txt on an old Mac; exact
# pins there would make the two impossible to satisfy at once. This server holds
# other people's Garmin tokens, so here an unreviewed rebuild picking up a new
# release on its own is a risk rather than a convenience.
#
# To move these: bump, run tests/hosted_test.py, and deploy deliberately.
mcp==2.2.0
garminconnect==0.3.16

# Direct dependency, not just a transitive one: store.py encrypts every stored
# token with Fernet. It arrived via garminconnect before, which meant the
# encryption library was only ever pinned by accident.
cryptography==50.0.1

uvicorn==0.54.0

# get_plan_chart draws the plan as a PNG for the conversation. chart.py is
# imported inside that tool rather than at module load, so the server starts
# without this — but the tool is advertised, so it has to work when called.
pillow==12.3.0

# Garmin's official API offers each activity's FIT file; official_fit.py reads
# the session's figures from it (heart rate zones, running dynamics, training
# effect, power) and keeps only those.
fitdecode==0.11.0

# Reads the country table (geo.py): which country someone signs up from.
maxminddb==2.8.2
